Citas: idCita(PK), fecha, hora, idPacientes(FK), estado(marca si se cancela), idTratamiento(FK), anotaciones
Pacientes: idPacientes(PK), nombre, apellido1, apellido2, email, telefono, fechaNacim, sexo, dni
Admin: idAdmin(PK), nombre, email, telefono
Promociones: idPromocion(PK), descripcion, fechaInicio, fechaFin, descuento
Tratamientos: idTratamiento(PK), nombre, descripcion, precio, fechaInicio, fechaFin, estado(en curso o finalizado)
Historial: idHistorial(PK), idPacientes(FK), fecha, descripcion
Cancelacion: idCancenlacion, idCita, vista

menu:

inicio -> login 

citas -> calendario - tramos - proximas citas - Historial
